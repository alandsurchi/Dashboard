String getTimeString() {
  DateTime now = rtc.now();
  int h = now.hour();
  String ampm = (h >= 12) ? "PM" : "AM";
  h = h % 12; if (h == 0) h = 12; 
  char buf[30];
  sprintf(buf, "%02d:%02d:%02d %s", h, now.minute(), now.second(), ampm.c_str());
  return String(buf);
}

void updateLCD() {
  DateTime now = rtc.now();
  float t = dht.readTemperature();
  lcd.setCursor(0, 0);
  if (isNightMode) lcd.print("MOON: NIGHT MODE"); 
  else if (t > tempThreshold) lcd.print("HOT! FAN ACTIVE "); 
  else if (motionActive) lcd.print("!! MOTION !!    "); 
  else {
    int h12 = now.hour() % 12;
    if (h12 == 0) h12 = 12;
    String ampm = (now.hour() >= 12) ? "PM" : "AM";
    char timeBuf[20];
    sprintf(timeBuf, "T:%02d:%02d%s D:%s", h12, now.minute(), ampm.c_str(), isDoorOpen ? "OPN" : "CLS");
    lcd.print(timeBuf);
  }
  lcd.setCursor(0, 1);
  if(isDoorOpen) lcd.print(" ACCESS GRANTED ");
  else lcd.print("Temp: " + String(t, 0) + "C       ");
}

void loop() {
  server.handleClient();
  
  // 1. FAN LOGIC
  float t = dht.readTemperature();
  if (t > tempThreshold || fanManualOverride) {
    digitalWrite(FAN_LED_PIN, HIGH);
    digitalWrite(REAL_FAN_PIN, HIGH);
  } else {
    digitalWrite(FAN_LED_PIN, LOW);
    digitalWrite(REAL_FAN_PIN, LOW);
  }

  // 2. NIGHT MODE (Affects L1, L2, AND L3)
  int lightVal = analogRead(LDR_PIN); 
  if (lightVal < 1000) { 
    isNightMode = true; 
    digitalWrite(LIGHT1_LED, HIGH); 
    digitalWrite(LIGHT2_LED, HIGH); 
    digitalWrite(LIGHT3_LED, HIGH); // <--- L3 Added
  } else if (isNightMode) {
      // Turn them all OFF when night ends
      digitalWrite(LIGHT1_LED, LOW); 
      digitalWrite(LIGHT2_LED, LOW); 
      digitalWrite(LIGHT3_LED, LOW); // <--- L3 Added
      isNightMode = false;
  }

  // 3. MOTION MODE (Affects L1, L2, AND L3)
  if (!isNightMode) {
    if (digitalRead(PIR_PIN) == HIGH) {
      if (!motionActive) { 
        motionActive = true; 
        digitalWrite(LIGHT1_LED, HIGH); 
        digitalWrite(LIGHT2_LED, HIGH); 
        digitalWrite(LIGHT3_LED, HIGH); // <--- L3 Added
      }
      motionTimer = millis();
    } else if (motionActive && (millis() - motionTimer > 5000)) {
         motionActive = false; 
         digitalWrite(LIGHT1_LED, LOW); 
         digitalWrite(LIGHT2_LED, LOW);
         digitalWrite(LIGHT3_LED, LOW); // <--- L3 Added
    }
  }

  static unsigned long lastLCD = 0;
  if (millis() - lastLCD > 1000) { lastLCD = millis(); updateLCD(); }

  if (isDoorOpen && millis() - doorTimer > 4000) { isDoorOpen = false; doorStatus = "Closed"; moveDoor(0); }

  if ( ! rfid.PICC_IsNewCardPresent()) return;
  if ( ! rfid.PICC_ReadCardSerial()) return;
  String content = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    content.concat(String(rfid.uid.uidByte[i] < 0x10 ? " 0" : " "));
    content.concat(String(rfid.uid.uidByte[i], HEX));
  }
  content.toUpperCase();
  if (content.substring(1) == authorizedUID) {
    playSuccessSound(); isDoorOpen = true; doorStatus = "Open"; doorTimer = millis(); moveDoor(180);
  } else {
    playErrorSound(); lcd.setCursor(0, 1); lcd.print(" ACCESS DENIED! "); delay(2000); 
  }
  rfid.PICC_HaltA(); rfid.PCD_StopCrypto1();
}

int getDistance() {
  digitalWrite(TRIG_PIN, LOW); delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH); delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH);
  int dist = duration * 0.034 / 2; 
  if (dist == 0 || dist > 400) return 400; 
  return dist;
}

// Optimized move function
void moveDoor(bool open) {
  if (open && !garageIsOpen) {
    Serial.println("ACTION: Opening Door");
    garageServo.attach(SERVO_PIN);
    garageServo.write(DOOR_OPEN_ANGLE);
    delay(500); garageServo.detach(); 
    garageIsOpen = true;
  } 
  else if (!open && garageIsOpen) {
    Serial.println("ACTION: Closing Door");
    garageServo.attach(SERVO_PIN);
    garageServo.write(DOOR_CLOSE_ANGLE);
    delay(500); garageServo.detach(); 
    garageIsOpen = false;
  }
}

void loop() {
  server.handleClient();
  currentDistance = getDistance();
  
  // 1. PHYSICAL BUTTON TOGGLE
  if (digitalRead(BUTTON_PIN) == LOW) {
    moveDoor(!garageIsOpen); // Toggle state: if closed, open. if open, close.
    delay(1000); // Debounce delay
  }

  // 2. AUTO-OPEN LOGIC (Only runs if door is currently CLOSED)
  if (!garageIsOpen) {
    bool carArriving = (deviceConnected && currentDistance < DISTANCE_THRESHOLD);
    if (carArriving) {
      if (detectionStartTime == 0) detectionStartTime = millis(); 
      else if (millis() - detectionStartTime > CONFIRMATION_DELAY) {
        moveDoor(true);
        detectionStartTime = 0;
      }
    } else {
      detectionStartTime = 0; 
    }
  }
  // Note: There is NO Auto-Close logic here. 
  // The door stays open until Case #1 (Button) or Web Server commands it to close.

  delay(100);
}

// 5. HANDLERS (unchanged)
// ---------------------------------------------------
void handleRoot() { server.send(200, "text/html", getHTML()); }
void handleSet() { 
  if (server.hasArg("val")) smokeThreshold = server.arg("val").toInt(); 
  server.send(200, "text/html", getHTML()); 
}
void handleEnable() { 
  systemEnabled = true; 
  server.send(200, "text/html", getHTML()); 
}
void handleStop() {
  systemEnabled = false;
  firePumpActive = false;
  digitalWrite(firePumpPin, HIGH);  // OFF
  digitalWrite(soilPumpPin, HIGH);  // OFF
  server.send(200, "text/html", getHTML());
}

// ---------------------------------------------------
// 6. SETUP (unchanged)
// ---------------------------------------------------
void setup() {
  Serial.begin(115200);

  pinMode(firePumpPin, OUTPUT);
  pinMode(soilPumpPin, OUTPUT);
  digitalWrite(firePumpPin, HIGH);  // OFF (active-low relay)
  digitalWrite(soilPumpPin, HIGH);  // OFF

  // Improve ADC accuracy
  analogSetPinAttenuation(mq2Pin, ADC_11db);
  analogSetPinAttenuation(soilSensorPin, ADC_11db);

  WiFi.begin(ssid, password);
  while (WiFi.status() != WL_CONNECTED) { delay(500); Serial.print("."); }
  Serial.println("\nWiFi Connected: " + WiFi.localIP().toString());

  server.on("/", handleRoot);
  server.on("/set", handleSet);
  server.on("/enable", handleEnable);
  server.on("/stop", handleStop);
  server.begin();
}

// ---------------------------------------------------
// 7. LOOP (unchanged)
// ---------------------------------------------------
void loop() {
  server.handleClient();

  // Emergency stop → force everything off
  if (!systemEnabled) {
    digitalWrite(firePumpPin, HIGH);
    digitalWrite(soilPumpPin, HIGH);
    firePumpActive = false;
    return;
  }

  // Read averaged values (reduces noise → fixes false triggers)
  int smokeValue = readAveraged(mq2Pin);
  int soilValue = readAveraged(soilSensorPin);

  // --- FIRE PUMP LOGIC (independent, only smoke sensor) ---
  if (smokeValue > smokeThreshold && !firePumpActive) {
    Serial.println("🔥 SMOKE DETECTED! Fire pump ON for 10s. Level: " + String(smokeValue));
    digitalWrite(firePumpPin, LOW);  // ON
    firePumpActive = true;
    fireTurnOffTime = millis() + fireTimerDuration;
  }
  if (firePumpActive && millis() >= fireTurnOffTime) {
    digitalWrite(firePumpPin, HIGH);  // OFF
    firePumpActive = false;
    Serial.println("Fire pump timed off.");
  }

  // --- SOIL PUMP LOGIC (independent, only soil sensor) ---
  static bool soilPumpOn = false;
  bool shouldWater = (soilValue > soilLimitDry && soilValue <= soilLimitNoSoil);

  if (shouldWater) {
    digitalWrite(soilPumpPin, LOW);  // ON
    if (!soilPumpOn) {
      Serial.println("💧 Dry soil → Soil pump ON. Level: " + String(soilValue));
      soilPumpOn = true;
    }
  } else {
    digitalWrite(soilPumpPin, HIGH); // OFF
    if (soilPumpOn) {
      String reason = (soilValue > soilLimitNoSoil) ? "NO SOIL" : "WET ENOUGH";
      Serial.println("Soil pump OFF (" + reason + "). Level: " + String(soilValue));
      soilPumpOn = false;
    }
  }
}
